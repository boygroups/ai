package mypack;

public class PrimeCheck {

    public String checkPrime(int n) {
        if(n <= 1) return "Not Prime";

        for(int i=2; i<=n/2; i++){
            if(n % i == 0)
                return "Not Prime";
        }
        return "Prime";
    }
}
